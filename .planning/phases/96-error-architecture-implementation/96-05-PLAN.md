       ```rust
       use libreroaster::error::app_error::{AppError, TemperatureError};
       use libreroaster::hardware::test_mocks::{MockThermometer, MockSsr};
       use libreroaster::hardware::max31856::Max31856Error;
       use libreroaster::hardware::ssr::SsrError;

       #[test]
       fn test_error_propagation_from_hardware_to_app() {
           // Mock sensor returns error
           let mut mock_sensor = MockThermometer::new();
           mock_sensor.inject_error(Max31856Error::CommunicationError);

           // Call sensor read (should fail)
           let result = mock_sensor.read_blocking();
           assert!(result.is_err());

           // Convert to AppError
           let app_err = AppError::from(result.unwrap_err());
           assert!(matches!(app_err, AppError::Control { .. })); // RoasterError then AppError
       }

       #[test]
       fn test_error_propagation_with_source_chaining() {
           let mut mock_ssr = MockSsr::new();
           mock_ssr.inject_error(SsrError::GpioError { source: "pin_config_failed" });

           let result = mock_ssr.set_duty(50);
           assert!(result.is_err());

           let app_err = AppError::from(result.unwrap_err());
           assert!(matches!(app_err, AppError::Hardware { .. }));
           assert_eq!(app_err.source(), Some("ssr_error"));
       }

       #[test]
       fn test_error_recovery_strategies() {
           // Timeout error should be recoverable
           let timeout_err = AppError::Communication {
               source: libreroaster::error::app_error::CommunicationError::TimeoutError,
           };
           assert!(timeout_err.is_recoverable());

           // Init error should not be recoverable
           let init_err = AppError::Initialization {
               source: libreroaster::error::app_error::InitError::HardwareInit {
                   what: "SPI",
                   reason: "timeout".to_string()
               },
           };
           assert!(!init_err.is_recoverable());
       }

       #[test]
       fn test_boundary_contract_conversions() {
           // Hardware -> Control -> App chain
           let sensor_err = Max31856Error::FaultDetected;
           let roaster_err = libreroaster::control::RoasterError::from(sensor_err);
           let app_err = AppError::from(roaster_err);

           assert!(matches!(app_err, AppError::Temperature { .. }));
           assert_eq!(app_err.source(), Some("sensor_fault"));
       }
       ```

    Why: Integration tests verify error propagation works correctly across all layer boundaries and conversion chains are valid.
