use embassy_time::Duration;
use critical_section::Mutex;

struct MockDriver;

unsafe impl Driver for MockDriver {
    fn now() -> u64 {
        // Return monotonic time in millis
        std::time::SystemTime::now()
            .duration_since(std::time::UNIX_EPOCH)
            .unwrap()
            .as_millis() as u64
    }
