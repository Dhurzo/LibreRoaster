  <files>internalDoc/ARCHITECTURE.md, internalDoc/PROTOCOL.md</files>
  <action>
    Extract all markdown links from both files:
    ```bash
    grep -oE '\[[^\]]+\]\([^)]+\)' internalDoc/ARCHITECTURE.md internalDoc/PROTOCOL.md
    ```

    Verify each internal link resolves to an existing file:
    ```bash
    for md_file in internalDoc/ARCHITECTURE.md internalDoc/PROTOCOL.md; do
        echo "Checking $md_file..."
        grep -oE '\[[^\]]+\]\(([^)]+)\)' "$md_file" | sed 's/.*(\([^)]*\).*/\1/' | while read link; do
            if [[ $link == http* ]]; then
                continue  # Skip external links
            fi
            if [[ $link == ../* ]]; then
                target="${link#../}"
            elif [[ $link == internalDoc/* ]]; then
                target="$link"
            else
                target="$(dirname "$md_file")/$link"
            fi
            if [ ! -f "$target" ]; then
                echo "  ❌ BROKEN LINK: $link"
            fi
        done
    done
    ```

    Document any broken links found.
  </action>
  <verify>All internal markdown links resolve to existing files</verify>
  <done>No broken cross-references</done>
